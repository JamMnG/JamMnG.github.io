⚙️ 핵심 로직 (접근성 전환 및 데이터 영속성)

UI 변수를 동적으로 제어하고 사용자 설정을 브라우저에 유지하기 위해 Vanilla JS와 LocalStorage를 활용한 핵심 로직입니다. 


// 1. 큰 글씨 모드(Senior Mode) 상태 전환 및 영속화
function applySeniorMode(isSenior) {
  // body 클래스 토글을 통해 CSS 변수 테마를 일괄 변경
  document.body.classList.toggle("senior-mode", isSenior);
  
  // 2. 사용자 설정을 브라우저 LocalStorage에 저장
  localStorage.setItem("dingdong:seniorMode", JSON.stringify(isSenior));
  
  // 버튼 접근성 상태 및 텍스트 동적 업데이트
  const toggleBtn = document.querySelector("#seniorToggle");
  toggleBtn.setAttribute("aria-pressed", isSenior);
  toggleBtn.textContent = isSenior ? "일반 모드" : "큰 글씨 모드";
}

// 3. AI 비전 자동 입력 로직 (편의성 강화)
function mockVisionAutofill() {
  // 이미지에서 추출된 데이터를 폼에 순차적으로 자동 기입

  document.querySelector("#itemPrice").value = "22000";
